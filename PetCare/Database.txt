USE master;

-- 1. Xóa database cũ để làm mới
IF EXISTS (SELECT * FROM sys.databases WHERE name = 'PetShopDB')
BEGIN
    ALTER DATABASE PetShopDB SET SINGLE_USER WITH ROLLBACK IMMEDIATE;
    DROP DATABASE PetShopDB;
END
GO

-- 2. Tạo lại Database
CREATE DATABASE PetShopDB;
GO

USE PetShopDB;
GO

-- 3. Tạo bảng Users (ĐÃ CÓ CẢ PHONE VÀ EMAIL)
CREATE TABLE Users (
    username NVARCHAR(50) PRIMARY KEY,
    password NVARCHAR(100) NOT NULL,
    role NVARCHAR(20) NOT NULL CHECK (role IN ('ADMIN', 'CUSTOMER')),
    fullName NVARCHAR(100) NOT NULL,
    Phone NVARCHAR(20) NULL,    -- Đã có từ lần trước
    Email NVARCHAR(100) NULL,   -- <--- MỚI THÊM CỘT NÀY ĐỂ SỬA LỖI
    createdAt DATETIME DEFAULT GETDATE()
);

-- 4. Tạo bảng Services
CREATE TABLE Services (
    id INT PRIMARY KEY IDENTITY(1,1),
    name NVARCHAR(200) NOT NULL,
    price DECIMAL(18,2) NOT NULL,
    isActive BIT DEFAULT 1,
    createdAt DATETIME DEFAULT GETDATE()
);

-- 5. Tạo bảng Pets
CREATE TABLE Pets (
    id INT PRIMARY KEY IDENTITY(1,1),
    name NVARCHAR(100) NOT NULL,
    species NVARCHAR(100) NOT NULL,
    age INT NOT NULL,
    ownerUsername NVARCHAR(50) NOT NULL,
    createdAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (ownerUsername) REFERENCES Users(username) ON DELETE CASCADE
);

-- 6. Tạo bảng Bookings
CREATE TABLE Bookings (
    id INT PRIMARY KEY IDENTITY(1,1),
    customerUser NVARCHAR(50) NOT NULL,
    petName NVARCHAR(100) NOT NULL,
    serviceName NVARCHAR(200) NOT NULL,
    price DECIMAL(18,2) NOT NULL,
    status NVARCHAR(50) NOT NULL CHECK (status IN (N'Chờ duyệt', N'Đã Duyệt', N'Đã Xong', N'Đã Hủy')),
    appointmentDate NVARCHAR(100) NOT NULL,
    createdAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (customerUser) REFERENCES Users(username)
);
